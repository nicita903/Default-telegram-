cmake_minimum_required(VERSION 3.22.1)
project(default_calls LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_NINJA_FORCE_RESPONSE_FILE ON CACHE BOOL "Use response files on Windows" FORCE)
get_filename_component(DEFAULT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(TGX_ROOT "${DEFAULT_ROOT}/.native/telegram-x-calls")
set(THIRDPARTY_DIR "${TGX_ROOT}/app/jni/third_party")
set(OPENSSL_ROOT "${DEFAULT_ROOT}/.native/openssl-install")
foreach(lib crypto ssl)
  add_library(${lib} STATIC IMPORTED GLOBAL)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${OPENSSL_ROOT}/lib/lib${lib}.a" INTERFACE_INCLUDE_DIRECTORIES "${OPENSSL_ROOT}/include")
endforeach()
set(OPUS_DIR "${THIRDPARTY_DIR}/opus")
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
include("${TGX_ROOT}/app/jni/BuildOpus.cmake")
set(YUV_DIR "${THIRDPARTY_DIR}/libyuv")
add_subdirectory("${YUV_DIR}" third_party/yuv EXCLUDE_FROM_ALL)
target_include_directories(yuv PUBLIC "${YUV_DIR}/include")
add_library(vpx STATIC IMPORTED GLOBAL)
set_target_properties(vpx PROPERTIES IMPORTED_LOCATION "${DEFAULT_ROOT}/.native/calls-vpx/libvpx.a"
  INTERFACE_INCLUDE_DIRECTORIES "${THIRDPARTY_DIR}/libvpx;${DEFAULT_ROOT}/.native/calls-vpx")
# Software H264 decoding is not enabled by this voice-only build; no FFmpeg binary is linked.
add_library(avcodec INTERFACE)
add_library(jni-utils INTERFACE)
add_library(cpufeatures STATIC "${ANDROID_NDK}/sources/android/cpufeatures/cpu-features.c")
target_include_directories(cpufeatures PUBLIC "${ANDROID_NDK}/sources/android/cpufeatures")
set(COMPAT_INCLUDE "${CMAKE_BINARY_DIR}/compat")
file(COPY "${OPUS_DIR}/include" DESTINATION "${COMPAT_INCLUDE}/third_party/opus/src")
file(COPY "${YUV_DIR}/include" DESTINATION "${COMPAT_INCLUDE}/third_party/libyuv")
file(COPY "${THIRDPARTY_DIR}/libvpx/vpx" DESTINATION "${COMPAT_INCLUDE}/third_party/libvpx/source/libvpx")
include_directories("${COMPAT_INCLUDE}" "${TGX_ROOT}/app/jni/tgvoip/third_party")
add_subdirectory("${TGX_ROOT}/app/jni/tgvoip" tgvoip EXCLUDE_FROM_ALL)
set_property(TARGET tgcalls PROPERTY CXX_STANDARD 20)
target_compile_options(tgcalls PRIVATE -Wno-deprecated-this-capture -Wno-error=reorder-ctor -Wno-error=unused-lambda-capture -Wno-error=unused-private-field)
# The disabled FFmpeg software decoder's color-space helper is also omitted.
get_target_property(WEBRTC_SOURCES webrtc SOURCES)
list(FILTER WEBRTC_SOURCES EXCLUDE REGEX "h264_color_space\\.cc$")
set_property(TARGET webrtc PROPERTY SOURCES ${WEBRTC_SOURCES})
get_target_property(ANDROID_SDK_SOURCES tgcallsjni SOURCES)
list(FILTER ANDROID_SDK_SOURCES INCLUDE REGEX "^.+/sdk/android/.*\\.cc$")
add_library(default_calls SHARED bridge.cpp ${ANDROID_SDK_SOURCES})
set_property(TARGET default_calls PROPERTY CXX_STANDARD 20)
target_include_directories(default_calls PRIVATE "${TGX_ROOT}/app/jni/tgvoip/third_party/tgcalls/tgcalls/third-party" "${WEBRTC_DIR}" "${TGX_ROOT}/app/jni/tgvoip/third_party/webrtc"
  "${TGX_ROOT}/app/jni/tgvoip/third_party/webrtc/generated" "${TGX_ROOT}/app/jni/tgvoip/third_party"
  "${TGX_ROOT}/app/jni/tgvoip/third_party/abseil-cpp" "${TGX_ROOT}/app/jni/tgvoip/third_party/webrtc_deps"
  "${TGX_ROOT}/app/jni/tgvoip/stub")
target_include_directories(webrtc PRIVATE "${TGX_ROOT}/app/jni/tgvoip/third_party/webrtc/generated" "${TGX_ROOT}/app/jni/tgvoip/third_party/libevent")
target_link_libraries(default_calls PRIVATE tgcalls webrtc cpufeatures log android EGL GLESv2 OpenSLES)
target_link_options(default_calls PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
